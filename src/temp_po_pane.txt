      {/* PO Details Pane - Right Side Slide-over */}
      {showPODetailsPane && selectedPOForDetails && createPortal(
        <>
          {/* Backdrop */}
          <div 
            className="fixed inset-0 bg-black/20 backdrop-blur-[2px] z-[110]" 
            onClick={() => {
              setShowPODetailsPane(false);
              setSelectedPOForDetails(null);
            }}
          />
          
          {/* Pane */}
          <div className="fixed right-0 top-0 bottom-0 w-[500px] bg-white shadow-2xl z-[120] flex flex-col">
            {/* Header */}
            <div className="px-6 py-4 border-b border-gray-200 flex items-center justify-between bg-gray-50">
              <div>
                <h3 className="text-base font-medium">Purchase Order Details</h3>
                <p className="text-xs text-gray-500 mt-0.5">
                  View order information and items
                </p>
              </div>
              <CloseButton
                onClick={() => {
                  setShowPODetailsPane(false);
                  setSelectedPOForDetails(null);
                }}
                variant="dark"
              />
            </div>

            {/* Content */}
            <div className="flex-1 overflow-y-auto p-6">
              {/* Order Summary */}
              <div className="space-y-6">
                <div>
                  <h4 className="text-sm font-medium mb-3 pb-2 border-b border-gray-200">Order Summary</h4>
                  <div className="space-y-3">
                    <div>
                      <label className="block text-xs text-gray-500 mb-1">PO Number</label>
                      <p className="text-sm text-gray-900 font-medium">{selectedPOForDetails.referenceNo}</p>
                    </div>
                    <div>
                      <label className="block text-xs text-gray-500 mb-1">Vendor</label>
                      <p className="text-sm text-gray-900">{selectedPOForDetails.vendor}</p>
                    </div>
                    <div>
                      <label className="block text-xs text-gray-500 mb-1">Warehouse</label>
                      <p className="text-sm text-gray-900">{selectedPOForDetails.destinationWarehouse}</p>
                    </div>
                    <div>
                      <label className="block text-xs text-gray-500 mb-1">Ordered Date</label>
                      <p className="text-sm text-gray-900">
                        {selectedPOForDetails.orderedDate 
                          ? new Date(selectedPOForDetails.orderedDate).toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' })
                          : '-'}
                      </p>
                    </div>
                    <div>
                      <label className="block text-xs text-gray-500 mb-1">Expected Date</label>
                      <p className="text-sm text-gray-900">
                        {new Date(selectedPOForDetails.expectedDate).toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' })}
                      </p>
                    </div>
                    <div>
                      <label className="block text-xs text-gray-500 mb-1">Status</label>
                      <span className={`inline-block px-2 py-1 text-xs rounded border ${
                        selectedPOForDetails.status === 'Received' 
                          ? 'bg-green-50 text-green-700 border-green-200'
                          : selectedPOForDetails.status === 'In Transit'
                          ? 'bg-blue-50 text-blue-700 border-blue-200'
                          : selectedPOForDetails.status === 'Overdue'
                          ? 'bg-red-50 text-red-700 border-red-200'
                          : 'bg-gray-50 text-gray-700 border-gray-200'
                      }`}>
                        {selectedPOForDetails.status}
                      </span>
                    </div>
                  </div>
                </div>

                {/* Order Items */}
                <div>
                  <h4 className="text-sm font-medium mb-3 pb-2 border-b border-gray-200">Order Items</h4>
                  <div className="space-y-2">
                    <div className="flex items-center justify-between py-2 px-3 bg-gray-50 rounded">
                      <span className="text-xs text-gray-600">Total Items</span>
                      <span className="text-sm text-gray-900 font-medium">{selectedPOForDetails.itemsCount}</span>
                    </div>
                    <div className="flex items-center justify-between py-2 px-3 bg-gray-50 rounded">
                      <span className="text-xs text-gray-600">Total Quantity</span>
                      <span className="text-sm text-gray-900 font-medium">{selectedPOForDetails.totalQuantity}</span>
                    </div>
                    <div className="flex items-center justify-between py-2 px-3 bg-gray-50 rounded">
                      <span className="text-xs text-gray-600">Received Quantity</span>
                      <span className="text-sm text-gray-900 font-medium">{selectedPOForDetails.receivedQuantity || 0}</span>
                    </div>
                  </div>
                </div>

                {/* Shipping Info */}
                <div>
                  <h4 className="text-sm font-medium mb-3 pb-2 border-b border-gray-200">Shipping Information</h4>
                  <div className="space-y-3">
                    <div>
                      <label className="block text-xs text-gray-500 mb-1">Carrier</label>
                      <p className="text-sm text-gray-900">{selectedPOForDetails.carrier || '-'}</p>
                    </div>
                    <div>
                      <label className="block text-xs text-gray-500 mb-1">Tracking Number</label>
                      <p className="text-sm text-gray-900 font-mono">{selectedPOForDetails.trackingNo || '-'}</p>
                    </div>
                    <div>
                      <label className="block text-xs text-gray-500 mb-1">Priority</label>
                      <span className={`inline-block px-2 py-1 text-xs rounded border ${
                        selectedPOForDetails.priority === 'High'
                          ? 'bg-red-50 text-red-700 border-red-200'
                          : selectedPOForDetails.priority === 'Medium'
                          ? 'bg-yellow-50 text-yellow-700 border-yellow-200'
                          : 'bg-gray-50 text-gray-700 border-gray-200'
                      }`}>
                        {selectedPOForDetails.priority}
                      </span>
                    </div>
                  </div>
                </div>
              </div>
            </div>

            {/* Footer Actions */}
            <div className="px-6 py-4 border-t border-gray-200 flex items-center justify-between bg-gray-50">
              <StyledButton
                variant="outline"
                size="medium"
                onClick={() => {
                  setShowPODetailsPane(false);
                  setSelectedPOForDetails(null);
                }}
              >
                Close
              </StyledButton>
              <div className="flex gap-3">
                <StyledButton
                  variant="outline"
                  size="medium"
                  onClick={() => {
                    console.log('Print PO:', selectedPOForDetails);
                    alert(`Print PO ${selectedPOForDetails.referenceNo}`);
                  }}
                >
                  <Printer className="w-4 h-4" />
                  <span>Print PO</span>
                </StyledButton>
                <StyledButton
                  variant="primary"
                  size="medium"
                  onClick={() => {
                    console.log('Receive Items:', selectedPOForDetails);
                    alert(`Receive items for ${selectedPOForDetails.referenceNo}`);
                  }}
                >
                  <Package className="w-4 h-4" />
                  <span>Receive Items</span>
                </StyledButton>
              </div>
            </div>
          </div>
        </>,
        document.body
      )}

    </ModulePageTemplate>
  );
}
